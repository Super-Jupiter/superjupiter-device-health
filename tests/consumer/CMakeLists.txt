cmake_minimum_required(VERSION 3.16)
project(SuperJupiterDeviceHealthConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_DEVICE_HEALTH_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_DEVICE_HEALTH_SOURCE_DIR}" device-health)
  if(TARGET superjupiter-device-health-tests OR TARGET superjupiter-device-health-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterDeviceHealth 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-device-health-consumer main.cpp)
target_link_libraries(superjupiter-device-health-consumer PRIVATE SuperJupiter::DeviceHealth)
